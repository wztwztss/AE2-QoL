```java
package com.wztwzt.ae2_qof.util;

import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

import net.minecraft.util.IIcon;
import net.minecraft.util.ResourceLocation;

import gregtech.api.enums.Textures;
import gregtech.api.interfaces.IIconContainer;
import gregtech.api.interfaces.ITexture;
import gregtech.api.render.TextureFactory;

import cpw.mods.fml.common.FMLCommonHandler;

/**
 * 自定义方块贴图工具：按资源路径加载 ae2_qof 自己的 16x16 方块贴图。
 *
 * <p><b>fix44 方案 P：GT 官方图标队列 + 资源包 PNG</b>——客户端用
 * gregtech.client.iconContainers.blocks.GTCustomBlockIconContainer（构造即入队
 * GregTechAPI.sGTBlockIconload，在 GT registerBlockIcons 阶段由
 * sBlockIcons.registerIcon 注册——该阶段位于 TextureStitchEvent.Pre 之前，
 * sprite 会正常参与 atlas stitch/UV 分配，与 GT++ 同路径）。PNG 由资源包
 * （AE2QoL-v7-resourcepack.zip，优先级高于 mod jar）提供，贴图资源与 mod 解耦。</p>
 */
public final class ModTextures {

    public static final String BASE = "ae2_qof:blocks/machine/";

    /** 全部机器基名（8 个），每个有 主体/_side/_top 三张贴图 + 共用 casing_side，共 25 张。 */
    public static final String[] MACHINES = {
        "universal_maintenance_hatch",
        "adaptive_net_terminal",
        "adaptive_net_hatch",
        "adaptive_net_laser_hatch",
        "adaptive_net_dynamo_hatch",
        "adaptive_net_laser_target",
        "wireless_energy_input",
        "wireless_energy_output"
    };

    private static final Map<String, ITexture> CACHE = new HashMap<>();
    private static final Map<String, IIcon> ICONS = new HashMap<>();
    private static final Set<String> MISS_LOGED = new HashSet<>();

    private ModTextures() {}

    public static void putIcon(String path, IIcon icon) { ICONS.put(path, icon); }
    public static int iconCount() { return ICONS.size(); }

    public static ITexture tex(String path) {
        ITexture t = CACHE.get(path);
        if (t == null) {
            if (FMLCommonHandler.instance().getEffectiveSide().isClient()) {
                IIconContainer gt =
                    gregtech.client.iconContainers.blocks.GTCustomBlockIconContainer.create(path);
                t = TextureFactory.of(new GtFallbackContainer(gt, path));
                System.out.println("[AE2QoL-TEX] tex() created GT-queue container for " + path);
            } else {
                t = TextureFactory.of(new ServerSafeContainer(path));
                System.out.println("[AE2QoL-TEX] tex() created server-safe container for " + path);
            }
            CACHE.put(path, t);
        }
        return t;
    }

    /** 资源包 PNG 是否可达（客户端缓存判定一次；服务端返回 true 无渲染意义）。 */
    private static Boolean ready = null;

    public static boolean isReady() {
        if (ready == null) {
            if (FMLCommonHandler.instance().getEffectiveSide().isClient()) {
                ready = checkResourceReachable(BASE + MACHINES[0]);
                System.out.println("[AE2QoL-TEX] isReady()=" + ready);
            } else {
                ready = true;
            }
        }
        return ready;
    }

    private static boolean checkResourceReachable(String path) {
        try {
            int colon = path.indexOf(':');
            if (colon < 0) { return false; }
            String domain = path.substring(0, colon);
            String pngPath = "textures/" + path.substring(colon + 1) + ".png";
            net.minecraft.client.resources.IResource res =
                net.minecraft.client.Minecraft.getMinecraft().getResourceManager()
                    .getResource(new net.minecraft.util.ResourceLocation(domain, pngPath));
            res.getInputStream().close();
            return true;
        } catch (Throwable t) {
            return false;
        }
    }

    /** 预创建全部 25 个贴图容器（含 GT 队列入队）。必须在 GT registerBlockIcons flush 之前调用（FMLInitialization）。 */
    public static void init() {
        if (!isReady()) {
            System.out.println("[AE2QoL-TEX] init() skipped: resource pack PNG not reachable (machines keep default GT look)");
            return;

                return Textures.BlockIcons.MACHINE_LV_SIDE.getIcon();

            if (MISS_LOGED.add(path)) {
```

```java
@SideOnly(Side.CLIENT)
public final class ClientTextureRegistry {

    public static final ClientTextureRegistry INSTANCE = new ClientTextureRegistry();
    private static boolean initialized = false;

    private ClientTextureRegistry() {}

    public static void init() {
        if (initialized) { return; }
        initialized = true;
        System.out.println("[AE2QoL-TEX] ClientTextureRegistry.init() registering instance on EVENT_BUS");
        MinecraftForge.EVENT_BUS.register(INSTANCE);
        System.out.println("[AE2QoL-TEX] registered instance. ready to receive TextureStitchEvent");
    }

    @SubscribeEvent
    public void onTextureStitch(TextureStitchEvent.Pre event) {
        System.out.println("[AE2QoL-TEX] TextureStitchEvent.Pre fired, textureType=" + event.map.getTextureType());
        if (event.map.getTextureType() != 0) { return; }
        for (String name : ModTextures.MACHINES) {
            register(event, ModTextures.BASE + name);
            register(event, ModTextures.BASE + name + "_top");
            register(event, ModTextures.BASE + name + "_side");
        }
        register(event, ModTextures.BASE + "casing_side");
        System.out.println("[AE2QoL-TEX] stitch complete, iconCount=" + ModTextures.iconCount());
    }

    private void register(TextureStitchEvent.Pre event, String path) {
        try {
            checkResourceReachable(path);
            IIcon icon = event.map.registerIcon(path);
            ModTextures.putIcon(path, icon);
            String name = icon == null ? "NULL-ICON" : icon.getIconName();
            int w = 0; int h = 0;
            try { w = icon.getIconWidth(); h = icon.getIconHeight(); } catch (Throwable ignored) {}
            System.out.println("[AE2QoL-TEX] registered " + path + " -> " + name + " size=" + w + "x" + h);
        } catch (Throwable t) {
            System.out.println("[AE2QoL-TEX] register FAILED for " + path + ": " + t);
        }
    }

    private static void checkResourceReachable(String path) {
        int colon = path.indexOf(':');
        if (colon < 0) { System.out.println("[AE2QoL-TEX] checkResource: bad path " + path); return; }
        String domain = path.substring(0, colon);
        String pngPath = "textures/" + path.substring(colon + 1) + ".png";
        ResourceLocation rl = new ResourceLocation(domain, pngPath);
        try {
            net.minecraft.client.resources.IResource res = Minecraft.getMinecraft().getResourceManager().getResource(rl);
            InputStream in = res.getInputStream();
            byte[] head = new byte[16];
            int n = in.read(head);
            in.close();
            StringBuilder hex = new StringBuilder();
            for (int i = 0; i < n; i++) { hex.append(String.format("%02X", head[i])); }
            System.out.println("[AE2QoL-TEX] resource OK " + rl + " head=" + hex);
        } catch (IOException e) {
            System.out.println("[AE2QoL-TEX] resource MISSING " + rl + " -> " + e);
        }
    }
}
```

### 4.3 日志判读（`[AE2QoL-TEX]`）

- `isReady()=true` + `tex() created GT-queue container`（启动阶段）→ 装包生效，v7 显示；
- `isReady()=false` + `init() skipped` → 资源包未启用，机器原样（符合预期）；
- `resource MISSING` → zip 路径或资源包未启用；
- `getIcon MISS ... fallback GT casing` → GT 队列失败，显示机箱（安全）。
